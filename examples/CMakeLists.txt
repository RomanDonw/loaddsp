macro(example name sourcefile)
    add_library(${name} SHARED "${CMAKE_CURRENT_SOURCE_DIR}/${sourcefile}")
    target_include_directories(${name} PRIVATE "${CMAKE_SOURCE_DIR}/include")
    set_target_properties(${name} PROPERTIES PREFIX "")
endmacro()
